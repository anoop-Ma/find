const App = {
  state: {
    theme: localStorage.getItem('tv-theme') || 'light',
    searchHistory: JSON.parse(localStorage.getItem('tv-search-history') || '[]'),
    bookmarks: JSON.parse(localStorage.getItem('tv-bookmarks') || '[]'),
    notes: JSON.parse(localStorage.getItem('tv-notes') || '[]'),
    recentSearches: JSON.parse(localStorage.getItem('tv-recent-searches') || '[]'),
    affiliateProducts: JSON.parse(localStorage.getItem('tv-affiliates') || '[]'),
    contentPlans: JSON.parse(localStorage.getItem('tv-plans') || '[]'),
    currentKeyword: '',
    currentCategory: 'all',
    toastQueue: []
  },

  init() {
    this.applyTheme();
    this.initEventListeners();
    this.loadTrends();
    this.loadCategories();
    this.loadStats();
    this.renderBookmarks();
    this.renderNotes();
    this.renderRecentSearches();
    document.body.style.display = 'block';
  },

  applyTheme() {
    document.documentElement.setAttribute('data-theme', this.state.theme);
    const icon = document.querySelector('.theme-toggle i');
    if (icon) icon.className = this.state.theme === 'dark' ? 'fas fa-sun' : 'fas fa-moon';
  },

  toggleTheme() {
    this.state.theme = this.state.theme === 'dark' ? 'light' : 'dark';
    localStorage.setItem('tv-theme', this.state.theme);
    this.applyTheme();
  },

  initEventListeners() {
    document.addEventListener('click', (e) => {
      const target = e.target.closest('[data-action]');
      if (!target) return;
      const action = target.dataset.action;
      if (typeof this[action] === 'function') this[action](e, target);
    });

    document.addEventListener('click', (e) => {
      const suggestions = document.getElementById('searchSuggestions');
      if (suggestions && !e.target.closest('.search-container')) {
        suggestions.classList.remove('show');
      }
    });

    const searchInput = document.getElementById('globalSearch');
    if (searchInput) {
      searchInput.addEventListener('input', this.debounce(() => this.handleSearchInput(), 300));
      searchInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') this.performSearch(searchInput.value);
      });
    }

    document.addEventListener('keydown', (e) => {
      if ((e.ctrlKey || e.metaKey) && e.key === 'k') {
        e.preventDefault();
        const input = document.getElementById('globalSearch');
        if (input) input.focus();
      }
    });
  },

  debounce(fn, delay) {
    let timer;
    return (...args) => {
      clearTimeout(timer);
      timer = setTimeout(() => fn(...args), delay);
    };
  },

  toast(message, type = 'info') {
    const container = document.getElementById('toastContainer');
    if (!container) return;
    const el = document.createElement('div');
    el.className = `toast-msg ${type}`;
    const icons = { success: 'fa-check-circle', error: 'fa-exclamation-circle', info: 'fa-info-circle' };
    el.innerHTML = `<i class="fas ${icons[type] || icons.info}"></i> ${message}`;
    container.appendChild(el);
    setTimeout(() => { el.style.opacity = '0'; setTimeout(() => el.remove(), 300); }, 3000);
  },

  async loadCategories() {
    try {
      const res = await fetch('data/categories.json');
      const categories = await res.json();
      const grid = document.getElementById('categoryGrid');
      if (!grid) return;
      grid.innerHTML = categories.map(c => `
        <div class="category-card glass-card" data-action="filterCategory" data-category="${c.id}" style="border-top: 3px solid ${c.color}">
          <div class="cat-icon" style="color:${c.color}"><i class="fas ${c.icon}"></i></div>
          <div class="cat-name">${c.name}</div>
        </div>
      `).join('');
    } catch (e) { console.warn('Categories load skipped'); }
  },

  async loadTrends() {
    try {
      const res = await fetch('data/mock-trends.json');
      const data = await res.json();
      this.renderTrendingKeywords(data.trendingKeywords);
      this.renderContentOpportunities(data.contentIdeas);
    } catch (e) { console.warn('Trends load skipped'); }
  },

  renderTrendingKeywords(keywords) {
    const container = document.getElementById('trendingKeywords');
    if (!container) return;
    container.innerHTML = keywords.map((k, i) => `
      <div class="trend-card glass-card fade-in" style="animation-delay:${i * 0.1}s" data-action="selectKeyword" data-keyword="${k.keyword}">
        <div class="trend-rank">#${i + 1}</div>
        <div class="trend-info">
          <div class="trend-keyword">${k.keyword}</div>
          <div class="trend-meta">${(k.volume / 1000).toFixed(0)}K searches/mo</div>
        </div>
        <span class="trend-badge ${k.trend}"><i class="fas fa-arrow-${k.trend === 'up' ? 'up' : k.trend === 'down' ? 'down' : 'right'}"></i> ${k.trend}</span>
      </div>
    `).join('');
  },

  renderContentOpportunities(ideas) {
    const container = document.getElementById('contentOpportunities');
    if (!container) return;
    container.innerHTML = ideas.map(i => `
      <div class="content-item d-flex align-items-center justify-content-between p-3 glass-card mb-2 fade-in" style="cursor:pointer" data-action="copyText" data-text="${i.title}">
        <div>
          <div style="font-weight:500;font-size:0.95rem">${i.title}</div>
          <span class="opportunity-badge ${i.difficulty === 'easy' ? 'high' : i.difficulty === 'medium' ? 'medium' : 'low'}">${i.difficulty}</span>
          <span class="opportunity-badge high"><i class="fas fa-fire"></i> ${i.viralPotential}% viral</span>
        </div>
        <button class="btn btn-sm btn-outline-custom"><i class="far fa-copy"></i></button>
      </div>
    `).join('');
  },

  loadStats() {
    const counters = document.querySelectorAll('.stat-number');
    counters.forEach(c => {
      const target = parseInt(c.dataset.count) || 0;
      let current = 0;
      const step = Math.ceil(target / 40);
      const interval = setInterval(() => {
        current += step;
        if (current >= target) { current = target; clearInterval(interval); }
        c.textContent = current.toLocaleString();
      }, 30);
    });
  },

  filterCategory(e, target) {
    const category = target.dataset.category;
    this.state.currentCategory = category;
    document.querySelectorAll('.category-card').forEach(c => c.style.opacity = category === 'all' || c.dataset.category === category ? '1' : '0.4');
    this.toast(`Showing ${target.querySelector('.cat-name')?.textContent || category} trends`, 'info');
  },

  handleSearchInput() {
    const input = document.getElementById('globalSearch');
    const suggestions = document.getElementById('searchSuggestions');
    if (!input || !suggestions) return;
    const val = input.value.trim();
    if (val.length < 2) { suggestions.classList.remove('show'); return; }
    const mock = ['trending', 'viral', 'how to', 'best', 'top 10', 'review', 'tutorial', 'guide', 'tips', 'ideas']
      .filter(s => s.includes(val.toLowerCase()));
    suggestions.innerHTML = mock.map(s => `<div class="suggestion-item" data-action="selectSuggestion" data-keyword="${s}"><i class="fas fa-search"></i> ${s}</div>`).join('');
    suggestions.classList.add('show');
  },

  selectSuggestion(e, target) {
    const keyword = target.dataset.keyword;
    document.getElementById('globalSearch').value = keyword;
    document.getElementById('searchSuggestions').classList.remove('show');
    this.performSearch(keyword);
  },

  selectKeyword(e, target) {
    const keyword = target.dataset.keyword;
    this.performSearch(keyword);
  },

  performSearch(keyword) {
    if (!keyword || keyword.trim().length < 1) return;
    keyword = keyword.trim();
    this.state.currentKeyword = keyword;
    const input = document.getElementById('globalSearch');
    if (input) input.value = keyword;
    document.getElementById('searchSuggestions')?.classList.remove('show');

    this.addToRecent(keyword);
    this.showSearchResults(keyword);
    this.toast(`Searching: "${keyword}"`, 'info');
  },

  addToRecent(keyword) {
    this.state.recentSearches = this.state.recentSearches.filter(s => s !== keyword);
    this.state.recentSearches.unshift(keyword);
    if (this.state.recentSearches.length > 20) this.state.recentSearches.pop();
    localStorage.setItem('tv-recent-searches', JSON.stringify(this.state.recentSearches));
    this.renderRecentSearches();
  },

  renderRecentSearches() {
    const container = document.getElementById('recentSearches');
    if (!container) return;
    if (this.state.recentSearches.length === 0) {
      container.innerHTML = '<div class="text-muted" style="font-size:0.85rem">No recent searches</div>';
      return;
    }
    container.innerHTML = this.state.recentSearches.map(s =>
      `<div class="sidebar-item" data-action="performSearch" data-keyword="${s}"><i class="fas fa-history"></i> ${s}</div>`
    ).join('');
  },

  copyText(e, target) {
    const text = target.dataset.text || target.closest('[data-text]')?.dataset.text;
    if (!text) return;
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(() => {
        this.toast('Copied to clipboard!', 'success');
      }).catch(() => {
        this.fallbackCopy(text);
      });
    } else {
      this.fallbackCopy(text);
    }
  },

  fallbackCopy(text) {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.position = 'fixed';
    ta.style.opacity = '0';
    document.body.appendChild(ta);
    ta.select();
    document.execCommand('copy');
    document.body.removeChild(ta);
    this.toast('Copied to clipboard!', 'success');
  },

  addBookmark(e, target) {
    const keyword = this.state.currentKeyword || target.dataset.keyword || 'untitled';
    if (this.state.bookmarks.some(b => b.keyword === keyword)) {
      this.toast('Already bookmarked!', 'error');
      return;
    }
    this.state.bookmarks.unshift({ keyword, date: new Date().toISOString(), notes: '' });
    localStorage.setItem('tv-bookmarks', JSON.stringify(this.state.bookmarks));
    this.renderBookmarks();
    this.toast(`Bookmarked "${keyword}"`, 'success');
  },

  removeBookmark(e, target) {
    const keyword = target.dataset.keyword;
    this.state.bookmarks = this.state.bookmarks.filter(b => b.keyword !== keyword);
    localStorage.setItem('tv-bookmarks', JSON.stringify(this.state.bookmarks));
    this.renderBookmarks();
    this.toast('Bookmark removed', 'info');
  },

  renderBookmarks() {
    const container = document.getElementById('bookmarksList');
    if (!container) return;
    if (this.state.bookmarks.length === 0) {
      container.innerHTML = '<div class="empty-state"><i class="fas fa-bookmark"></i><p>No bookmarks yet. Search for keywords and bookmark them!</p></div>';
      return;
    }
    container.innerHTML = this.state.bookmarks.map(b => `
      <div class="bookmark-item fade-in">
        <div>
          <div class="bookmark-title"><i class="fas fa-bookmark text-accent"></i> ${b.keyword}</div>
          <div class="bookmark-meta">${new Date(b.date).toLocaleDateString()}</div>
        </div>
        <div class="bookmark-actions">
          <button class="btn btn-sm btn-outline-custom" data-action="performSearch" data-keyword="${b.keyword}"><i class="fas fa-search"></i></button>
          <button class="btn btn-sm btn-danger-custom" data-action="removeBookmark" data-keyword="${b.keyword}"><i class="fas fa-trash"></i></button>
        </div>
      </div>
    `).join('');
  },

  saveNote() {
    const textarea = document.getElementById('noteEditor');
    if (!textarea) return;
    const text = textarea.value.trim();
    if (!text) { this.toast('Write something first!', 'error'); return; }
    this.state.notes.unshift({ text, date: new Date().toISOString(), id: Date.now() });
    localStorage.setItem('tv-notes', JSON.stringify(this.state.notes));
    textarea.value = '';
    this.renderNotes();
    this.toast('Note saved!', 'success');
  },

  deleteNote(e, target) {
    const id = parseInt(target.dataset.id);
    this.state.notes = this.state.notes.filter(n => n.id !== id);
    localStorage.setItem('tv-notes', JSON.stringify(this.state.notes));
    this.renderNotes();
    this.toast('Note deleted', 'info');
  },

  renderNotes() {
    const container = document.getElementById('notesList');
    if (!container) return;
    if (this.state.notes.length === 0) {
      container.innerHTML = '<div class="empty-state"><i class="fas fa-sticky-note"></i><p>No notes yet. Start writing!</p></div>';
      return;
    }
    container.innerHTML = this.state.notes.map(n => `
      <div class="bookmark-item fade-in">
        <div style="flex:1">
          <div style="font-size:0.9rem;white-space:pre-wrap">${this.escapeHtml(n.text)}</div>
          <div class="bookmark-meta mt-1">${new Date(n.date).toLocaleDateString()} ${new Date(n.date).toLocaleTimeString()}</div>
        </div>
        <button class="btn btn-sm btn-danger-custom" data-action="deleteNote" data-id="${n.id}"><i class="fas fa-trash"></i></button>
      </div>
    `).join('');
  },

  escapeHtml(text) {
    const d = document.createElement('div');
    d.textContent = text;
    return d.innerHTML;
  },

  renderCalendar(monthOffset = 0) {
    const container = document.getElementById('calendarGrid');
    if (!container) return;
    const now = new Date();
    const year = now.getFullYear();
    const month = now.getMonth() + monthOffset;
    const firstDay = new Date(year, month, 1).getDay();
    const daysInMonth = new Date(year, month + 1, 0).getDate();
    const today = now.getDate();

    const days = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
    let html = days.map(d => `<div class="cal-header">${d}</div>`).join('');
    for (let i = 0; i < firstDay; i++) html += '<div></div>';
    for (let d = 1; d <= daysInMonth; d++) {
      const isToday = d === today && monthOffset === 0;
      const hasEvent = this.state.contentPlans.some(p => {
        const pd = new Date(p.date);
        return pd.getDate() === d && pd.getMonth() === month && pd.getFullYear() === year;
      });
      html += `<div class="cal-day ${isToday ? 'today' : ''} ${hasEvent ? 'has-event' : ''}" title="${hasEvent ? 'Has planned content' : ''}">${d}</div>`;
    }
    container.innerHTML = html;
    const label = document.getElementById('calendarLabel');
    if (label) label.textContent = new Date(year, month).toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
  },

  exportBookmarks() {
    const data = JSON.stringify(this.state.bookmarks, null, 2);
    const blob = new Blob([data], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `trendverse-bookmarks-${new Date().toISOString().split('T')[0]}.json`;
    a.click();
    URL.revokeObjectURL(url);
    this.toast('Bookmarks exported!', 'success');
  },

  exportNotes() {
    const text = this.state.notes.map(n => `[${new Date(n.date).toLocaleString()}]\n${n.text}\n`).join('\n---\n\n');
    const blob = new Blob([text], { type: 'text/plain' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `trendverse-notes-${new Date().toISOString().split('T')[0]}.txt`;
    a.click();
    URL.revokeObjectURL(url);
    this.toast('Notes exported!', 'success');
  }
