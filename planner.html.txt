  <script>
    document.addEventListener('DOMContentLoaded', function() {
      App.renderCalendar(0);
      App.renderPlans();
      document.getElementById('planDate').valueAsDate = new Date();
    });

    function updateProgress() {
      const plans = App.state.contentPlans;
      const ideas = plans.filter(p => p.status === 'idea').length;
      const drafting = plans.filter(p => p.status === 'drafting' || p.status === 'scheduled').length;
      const published = plans.filter(p => p.status === 'published').length;
      const total = Math.max(plans.length, 1);
      document.getElementById('progressIdeas').textContent = ideas;
      document.getElementById('progressDrafting').textContent = drafting;
      document.getElementById('progressPublished').textContent = published;
      document.getElementById('progressIdeasBar').style.width = (ideas / total * 100) + '%';
      document.getElementById('progressDraftingBar').style.width = (drafting / total * 100) + '%';
      document.getElementById('progressPublishedBar').style.width = (published / total * 100) + '%';
    }

    const origRender = App.renderPlans;
    App.renderPlans = function() {
      origRender.call(App);
      updateProgress();
    };
  </script>
